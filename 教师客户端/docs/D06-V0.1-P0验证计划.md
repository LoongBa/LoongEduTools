```powershell
# 1. 拷贝整个目录到 U 盘
robocopy "F:\LoongBa_Git\LoongEduTools\教师客户端\src\src-tauri\target\release" "E:\TeacherClient" /E /NFL /NDL /NJH /NJS

# 2. 目标机（Win7/Win11）双击 E:\TeacherClient\teacher-client.exe
# 3. 验证：config.json 生成在 exe 同目录（而非 AppData）
```

### 5.3 WebView2 缺失专项

```powershell
# 1. 卸载 WebView2（控制面板 → 程序 → Microsoft Edge WebView2 Runtime）
# 2. 双击 teacher-client.exe
# 3. 预期：弹出原生对话框"首次使用需安装 WebView2 运行时（约 2 分钟，仅需一次）[去安装]/[退出]"
# 4. 点[去安装] → 启动同目录 MicrosoftEdgeWebview2Setup.exe → 完成后重启 exe 正常
```

### 5.4 内容包篡改测试

```powershell
# 1. 备份 manifest.json
Copy-Item packages-embedded\pep-reader-u01\manifest.json packages-embedded\pep-reader-u01\manifest.json.bak

# 2. 篡改一个字节（如 version 从 "1.3.3" 改成 "1.3.4"）
(Get-Content packages-embedded\pep-reader-u01\manifest.json -Raw) -replace '"1\.3\.3"','"1.3.4"' | Set-Content packages-embedded\pep-reader-u01\manifest.json

# 3. 启动 exe → 打开英语点读
# 4. 预期：签名校验失败 → 拒绝加载 + 错误提示（不崩溃）
# 5. 恢复备份
Move-Item packages-embedded\pep-reader-u01\manifest.json.bak packages-embedded\pep-reader-u01\manifest.json -Force
```
